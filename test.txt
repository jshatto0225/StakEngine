-- This works but logging must happen after the constructor of the layer
-- This can be easily expanded for other systems like Input
-- It would require any initialization that can produce an error message to happen in OnAttach
  -- Although that is probably fine and a good thing

class IAppLayer {
protected:
  void LogError();
  ...
private:
  friend class FApplication;
  SetLogger(Ref<FLogger> Logger);
}

class IAppLayerChild : public IAppLayer {

}

void AddLayer(IAppLayer *Layer, FBool CanLog) {
  if (CanLog) {
    Layer->SetLogger();
  }
  mLayerStack.push(Layer);
}

-- This allows logging in the constructor but it requires the
-- user know that the logger must be the first or last parameter

#define LOGGABLE_LAYER_CONSTRUCTOR(LayerType, ...) LayerType(Ref<FLogger> Logger, __VA_ARGS__) : mLogger(Logger)
class IAppLayer {
protected:
  template<typename ...TArgs>
  void LogError(TArgs&& ...Args) {
    mLogger->error(std::forward<TArgs>(Args)...);
  }
  void LogWarn() {
    ...
  }
  ...
private:
  Ref<FLogger> mLogger;
}

class IAppLayerChild : public IAppLayer {
  LOGGABLE_LAYER_CONSTRUCTOR(IAppLayerChild) {
  }
}

template<typename TLayerType, typename ...TArgs>
void TAddLayer(FBool CanLog, TArgs&& ...Args) {
  if (CanLog) {
    mLayers.Push(new TLayerType(mClientLogger, std::forward<TArgs>Aargs));
  }
  else {
    mLayers.Push(new TLayerType(std::forward<TArgs>Args));
  }
}
